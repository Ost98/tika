import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;
import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;
import java.io.Serializable;
import java.util.Comparator;

@ExtendWith(MockitoExtension.class)
public class Patterns_matches_5_1_Test {

    @Mock
    private MediaTypeRegistry registry;

    @Mock
    private MimeType mimeType;

    private Patterns patterns;

    @BeforeEach
    public void setUp() {
        patterns = new Patterns(registry);
    }

    @Test
    public void testMatchesExactMatch() {
        when(registry.getTypes()).thenReturn(new TreeSet<>());
        when(names.containsKey("text/plain")).thenReturn(true);
        when(names.get("text/plain")).thenReturn(mimeType);
        MimeType result = patterns.matches("text/plain");
        assertEquals(mimeType, result);
        verify(registry, never()).getTypes();
        verify(names, times(1)).containsKey("text/plain");
        verify(names, times(1)).get("text/plain");
    }

    @Test
    public void testMatchesExtensionMatch() {
        when(registry.getTypes()).thenReturn(new TreeSet<>());
        when(names.containsKey("text/plain")).thenReturn(false);
        when(extensions.containsKey(".txt")).thenReturn(true);
        when(extensions.get(".txt")).thenReturn(mimeType);
        MimeType result = patterns.matches("example.txt");
        assertEquals(mimeType, result);
        verify(registry, never()).getTypes();
        verify(names, times(1)).containsKey("example.txt");
        verify(extensions, times(1)).containsKey(".txt");
        verify(extensions, times(1)).get(".txt");
    }

    @Test
    public void testMatchesGlobMatch() {
        when(registry.getTypes()).thenReturn(new TreeSet<>());
        when(names.containsKey("text/plain")).thenReturn(false);
        when(extensions.containsKey(".txt")).thenReturn(false);
        when(globs.containsKey(".*\\.txt")).thenReturn(true);
        when(globs.get(".*\\.txt")).thenReturn(mimeType);
        MimeType result = patterns.matches("example.txt");
        assertEquals(mimeType, result);
        verify(registry, never()).getTypes();
        verify(names, times(1)).containsKey("example.txt");
        verify(extensions, times(1)).containsKey(".txt");
        verify(globs, times(1)).containsKey(".*\\.txt");
        verify(globs, times(1)).get(".*\\.txt");
    }

    @Test
    public void testMatchesNoMatch() {
        when(registry.getTypes()).thenReturn(new TreeSet<>());
        when(names.containsKey("text/plain")).thenReturn(false);
        when(extensions.containsKey(".txt")).thenReturn(false);
        when(globs.containsKey(".*\\.txt")).thenReturn(false);
        MimeType result = patterns.matches("example.txt");
        assertNull(result);
        verify(registry, never()).getTypes();
        verify(names, times(1)).containsKey("example.txt");
        verify(extensions, times(1)).containsKey(".txt");
        verify(globs, times(1)).containsKey(".*\\.txt");
    }
}
