package org.apache.tika.mime;

import org.apache.tika.mime.MediaType;
import org.apache.tika.mime.MediaTypeRegistry;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;
import java.io.Serializable;
import java.util.Comparator;
import java.util.HashMap;
import java.util.Map;
import java.util.SortedMap;
import java.util.TreeMap;
import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;

@ExtendWith(MockitoExtension.class)
public class Patterns_matches_5_1_Test {

    @Mock
    private MediaTypeRegistry registry;

    @Mock
    private MimeType mimeType;

    private Patterns patterns;

    @BeforeEach
    public void setUp() {
        patterns = new Patterns(registry);
        patterns.names = new HashMap<>();
        patterns.extensions = new HashMap<>();
        patterns.globs = new TreeMap<>(new LengthComparator());
    }

    @Test
    public void testMatchesExactMatch() {
        patterns.names.put("text/plain", mimeType);
        MimeType result = patterns.matches("text/plain");
        assertEquals(mimeType, result);
        verify(registry, never()).getTypes();
        verify(patterns.names, times(1)).containsKey("text/plain");
        verify(patterns.names, times(1)).get("text/plain");
    }

    @Test
    public void testMatchesExtensionMatch() {
        patterns.extensions.put(".txt", mimeType);
        MimeType result = patterns.matches("example.txt");
        assertEquals(mimeType, result);
        verify(registry, never()).getTypes();
        verify(patterns.names, times(1)).containsKey("example.txt");
        verify(patterns.extensions, times(1)).containsKey(".txt");
        verify(patterns.extensions, times(1)).get(".txt");
    }

    @Test
    public void testMatchesGlobMatch() {
        patterns.globs.put(".*\\.txt", mimeType);
        MimeType result = patterns.matches("example.txt");
        assertEquals(mimeType, result);
        verify(registry, never()).getTypes();
        verify(patterns.names, times(1)).containsKey("example.txt");
        verify(patterns.extensions, times(1)).containsKey(".txt");
        verify(patterns.globs, times(1)).containsKey(".*\\.txt");
        verify(patterns.globs, times(1)).get(".*\\.txt");
    }

    @Test
    public void testMatchesNoMatch() {
        MimeType result = patterns.matches("example.txt");
        assertNull(result);
        verify(registry, never()).getTypes();
        verify(patterns.names, times(1)).containsKey("example.txt");
        verify(patterns.extensions, times(1)).containsKey(".txt");
        verify(patterns.globs, times(1)).containsKey(".*\\.txt");
    }
}
