@ExtendWith(MockitoExtension.class)
public class Patterns_add_0_1_Test {

    @Mock
    private MediaTypeRegistry registry;

    @Mock
    private MimeType mimeType;

    private Patterns patterns;

    @BeforeEach
    public void setUp() {
        patterns = new Patterns(registry);
    }

    @Test
    public void testAdd() throws Exception {
        String pattern = "text/plain";
        when(mimeType.getType()).thenReturn(MediaType.parse("text/plain"));
        when(mimeType.getName()).thenReturn("text/plain");
        when(mimeType.getDescription()).thenReturn("Plain text");
        when(mimeType.getAcronym()).thenReturn("TXT");
        when(mimeType.getUniformTypeIdentifier()).thenReturn("public.plain-text");
        patterns.add(pattern, mimeType);
        verify(registry, times(1)).getDefaultRegistry();
        verify(mimeType, times(1)).getType();
        verify(mimeType, times(1)).getName();
        verify(mimeType, times(1)).getDescription();
        verify(mimeType, times(1)).getAcronym();
        verify(mimeType, times(1)).getUniformTypeIdentifier();
        verify(mimeType, times(1)).addExtension("txt");
        assertTrue(patterns.names.containsKey("text/plain"));
        assertTrue(patterns.extensions.containsKey("txt"));
        assertTrue(patterns.globs.containsKey("text/plain"));
    }

    @Test
    public void testAddWithJavaRegex() throws Exception {
        String pattern = "java\\.regex";
        when(mimeType.getType()).thenReturn(MediaType.parse("text/plain"));
        when(mimeType.getName()).thenReturn("text/plain");
        when(mimeType.getDescription()).thenReturn("Plain text");
        when(mimeType.getAcronym()).thenReturn("TXT");
        when(mimeType.getUniformTypeIdentifier()).thenReturn("public.plain-text");
        patterns.add(pattern, true, mimeType);
        verify(registry, times(1)).getDefaultRegistry();
        verify(mimeType, times(1)).getType();
        verify(mimeType, times(1)).getName();
        verify(mimeType, times(1)).getDescription();
        verify(mimeType, times(1)).getAcronym();
        verify(mimeType, times(1)).getUniformTypeIdentifier();
        verify(mimeType, times(1)).addExtension("java\\.regex");
        assertTrue(patterns.names.containsKey("java\\.regex"));
        assertTrue(patterns.extensions.containsKey("java\\.regex"));
        assertTrue(patterns.globs.containsKey("java\\.regex"));
    }

    @Test
    public void testAddWithExtension() throws Exception {
        String pattern = "*.txt";
        when(mimeType.getType()).thenReturn(MediaType.parse("text/plain"));
        when(mimeType.getName()).thenReturn("text/plain");
        when(mimeType.getDescription()).thenReturn("Plain text");
        when(mimeType.getAcronym()).thenReturn("TXT");
        when(mimeType.getUniformTypeIdentifier()).thenReturn("public.plain-text");
        patterns.add(pattern, mimeType);
        verify(registry, times(1)).getDefaultRegistry();
        verify(mimeType, times(1)).getType();
        verify(mimeType, times(1)).getName();
        verify(mimeType, times(1)).getDescription();
        verify(mimeType, times(1)).getAcronym();
        verify(mimeType, times(1)).getUniformTypeIdentifier();
        verify(mimeType, times(1)).addExtension("txt");
        assertTrue(patterns.names.containsKey("text/plain"));
        assertTrue(patterns.extensions.containsKey("txt"));
        assertTrue(patterns.globs.containsKey("text/plain"));
    }

    @Test
    public void testAddWithInvalidPattern() {
        assertThrows(IllegalArgumentException.class, () -> {
            patterns.add(null, mimeType);
        });
    }

    @Test
    public void testAddWithInvalidMimeType() {
        assertThrows(IllegalArgumentException.class, () -> {
            patterns.add("text/plain", null);
        });
    }
}
