public class Patterns_add_0_0_Test {

    private Patterns patterns;

    private MediaTypeRegistry registry;

    private MimeType mimeType;

    @BeforeEach
    public void setUp() {
        registry = mock(MediaTypeRegistry.class);
        mimeType = mock(MimeType.class);
        patterns = new Patterns(registry);
    }

    @Test
    public void testAddWithValidPatternAndMimeType() throws MimeTypeException {
        String pattern = "example.txt";
        when(mimeType.getType()).thenReturn(MediaType.parse("text/plain"));
        patterns.add(pattern, mimeType);
        verify(registry, times(1)).getDefaultRegistry();
        verify(mimeType, times(1)).addExtension(eq("txt"));
        verify(patterns, never()).addName(anyString(), any(MimeType.class));
        verify(patterns, never()).addExtension(anyString(), any(MimeType.class));
        verify(patterns, never()).addGlob(anyString(), anyBoolean(), any(MimeType.class));
    }

    @Test
    public void testAddWithNullPattern() {
        String pattern = null;
        assertThrows(IllegalArgumentException.class, () -> patterns.add(pattern, mimeType));
    }

    @Test
    public void testAddWithNullMimeType() {
        String pattern = "example.txt";
        assertThrows(IllegalArgumentException.class, () -> patterns.add(pattern, null));
    }

    @Test
    public void testAddWithJavaRegexPattern() throws MimeTypeException {
        String pattern = ".*\\.txt";
        when(mimeType.getType()).thenReturn(MediaType.parse("text/plain"));
        patterns.add(pattern, mimeType);
        verify(registry, times(1)).getDefaultRegistry();
        verify(mimeType, never()).addExtension(anyString());
        verify(patterns, never()).addName(anyString(), any(MimeType.class));
        verify(patterns, never()).addExtension(anyString(), any(MimeType.class));
        verify(patterns, times(1)).addGlob(eq(pattern), eq(mimeType));
    }

    @Test
    public void testAddWithNoWildcardPattern() throws MimeTypeException {
        String pattern = "example.txt";
        when(mimeType.getType()).thenReturn(MediaType.parse("text/plain"));
        patterns.add(pattern, mimeType);
        verify(registry, times(1)).getDefaultRegistry();
        verify(mimeType, times(1)).addExtension(eq("txt"));
        verify(patterns, times(1)).addName(eq(pattern), eq(mimeType));
        verify(patterns, never()).addExtension(anyString(), any(MimeType.class));
        verify(patterns, never()).addGlob(anyString(), anyBoolean(), any(MimeType.class));
    }

    @Test
    public void testAddWithSingleExtensionPattern() throws MimeTypeException {
        String pattern = "*.txt";
        when(mimeType.getType()).thenReturn(MediaType.parse("text/plain"));
        patterns.add(pattern, mimeType);
        verify(registry, times(1)).getDefaultRegistry();
        verify(mimeType, times(1)).addExtension(eq("txt"));
        verify(patterns, never()).addName(anyString(), any(MimeType.class));
        verify(patterns, times(1)).addExtension(eq("txt"), eq(mimeType));
        verify(patterns, never()).addGlob(anyString(), anyBoolean(), any(MimeType.class));
    }

    @Test
    public void testAddWithComplexPattern() throws MimeTypeException {
        String pattern = "[a-z]*.txt";
        when(mimeType.getType()).thenReturn(MediaType.parse("text/plain"));
        patterns.add(pattern, mimeType);
        verify(registry, times(1)).getDefaultRegistry();
        verify(mimeType, times(1)).addExtension(eq("txt"));
        verify(patterns, never()).addName(anyString(), any(MimeType.class));
        verify(patterns, never()).addExtension(anyString(), any(MimeType.class));
        verify(patterns, times(1)).addGlob(eq(pattern), eq(mimeType));
    }

    @Test
    public void testAddWithComplexPatternAndBoolean() throws MimeTypeException {
        String pattern = "[a-z]*.txt";
        when(mimeType.getType()).thenReturn(MediaType.parse("text/plain"));
        patterns.add(pattern, true, mimeType);
        verify(registry, times(1)).getDefaultRegistry();
        verify(mimeType, times(1)).addExtension(eq("txt"));
        verify(patterns, never()).addName(anyString(), any(MimeType.class));
        verify(patterns, never()).addExtension(anyString(), any(MimeType.class));
    }
}
